cmake_minimum_required(VERSION 3.13)
project(protocoldemo_bluez_trial LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
find_package(Threads REQUIRED)
set(PROTOCOL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../NeuroExoFirmware/lib/commProtocol/src")
add_executable(neuroexo_bluez_benchmark main.cpp "${PROTOCOL_DIR}/commProtocol.cpp")
target_include_directories(neuroexo_bluez_benchmark PRIVATE "${PROTOCOL_DIR}")
target_link_libraries(neuroexo_bluez_benchmark PRIVATE PkgConfig::GIO Threads::Threads)
target_compile_options(neuroexo_bluez_benchmark PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(protocoldemo_receiver_check test_receiver.cpp "${PROTOCOL_DIR}/commProtocol.cpp")
  target_include_directories(protocoldemo_receiver_check PRIVATE receiver_stubs "${PROTOCOL_DIR}")
  target_compile_options(protocoldemo_receiver_check PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
  add_test(NAME protocoldemo_receiver_check COMMAND protocoldemo_receiver_check)
  add_executable(protocoldemo_trial_check test_trial.cpp "${PROTOCOL_DIR}/commProtocol.cpp")
  target_include_directories(protocoldemo_trial_check PRIVATE "${PROTOCOL_DIR}")
  target_link_libraries(protocoldemo_trial_check PRIVATE Threads::Threads)
  target_compile_options(protocoldemo_trial_check PRIVATE -UNDEBUG)
  add_test(NAME protocoldemo_trial_check COMMAND protocoldemo_trial_check)
  add_executable(protocoldemo_bluez_check test_link.cpp "${PROTOCOL_DIR}/commProtocol.cpp")
  target_include_directories(protocoldemo_bluez_check PRIVATE "${PROTOCOL_DIR}")
  target_link_libraries(protocoldemo_bluez_check PRIVATE PkgConfig::GIO Threads::Threads)
  add_test(NAME protocoldemo_bluez_check COMMAND protocoldemo_bluez_check $<TARGET_FILE:neuroexo_bluez_benchmark>)
  set_tests_properties(protocoldemo_bluez_check PROPERTIES TIMEOUT 30)
endif()
